# Book Shelf 
















#version 300 es
precision highp float;

out vec4 O;

uniform float time;
uniform vec2 resolution;
uniform vec2 move;
uniform vec2 wheel;

#define FC gl_FragCoord.xy
#define R resolution
#define MN min(R.x, R.y)

vec3 g_col;
float g_glow;
float g_local_lit;

/* --------------------------------------------------------- */
/* Utility                                                    */
/* --------------------------------------------------------- */

float hash21(vec2 p)
{
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

float sdBox(vec3 p, vec3 b, float r)
{
    vec3 q = abs(p) - b + vec3(r);

    return length(max(q, vec3(0.0))) +
           min(max(q.x, max(q.y, q.z)), 0.0) - r;
}

vec3 bookColour(vec2 id, float variation)
{
    float c = hash21(id + vec2(4.7, 8.3));
    vec3 col;

    if (c < 0.18)
    {
        col = vec3(0.65, 0.12, 0.07);   // red
    }
    else if (c < 0.35)
    {
        col = vec3(0.12, 0.25, 0.16);   // green
    }
    else if (c < 0.50)
    {
        col = vec3(0.10, 0.18, 0.38);   // blue
    }
    else if (c < 0.70)
    {
        col = vec3(0.52, 0.30, 0.12);   // brown
    }
    else if (c < 0.84)
    {
        col = vec3(0.68, 0.48, 0.22);   // ochre
    }
    else
    {
        col = vec3(0.78, 0.72, 0.57);   // cream

    }

    col *= 0.82 + variation * 0.32;

    return col;
}

/* --------------------------------------------------------- */
/* Infinite bookshelf-building city                           */
/* --------------------------------------------------------- */

float map(vec3 p)
{
    g_col = vec3(0.05, 0.03, 0.015);
    g_glow = 0.0;
    g_local_lit = 0.0;

    /* Infinite city grid */
    float cellSize = 1.25;

    float row = floor(p.z / cellSize);
    float stagger = mod(row, 2.0) * cellSize * 0.5;

    vec2 id = vec2(
        floor((p.x + stagger) / cellSize),
        row
    );

    float localX = mod(p.x + stagger, cellSize) - cellSize * 0.5;
    float localZ = mod(p.z, cellSize) - cellSize * 0.5;

    float r1 = hash21(id);
    float r2 = hash21(id + vec2(12.3, 4.6));

    /* Each city object is a tall bookshelf tower */
    float W = 0.30 + r1 * 0.12;
    float D = 0.30 + r2 * 0.10;
    float H = 2.10 + hash21(id + vec2(8.4, 17.2)) * 2.10;

    /* Building starts at ground level */
    vec3 q = vec3(localX, p.y - H, localZ);

    float tower = sdBox(q, vec3(W, H, D), 0.012);

    /* Fast exit when far from a tower */
    if (tower > 0.14)
    {
        return tower;
    }

    float d = tower;

    vec3 darkWood = vec3(0.075, 0.034, 0.014);
    g_col = darkWood;

    /*
       Choose the closest vertical face.
       This puts bookshelf details on all four building sides.
    */
    vec3 f = q;
    float faceW = W;
    float faceD = D;

    if (abs(q.x) > abs(q.z))
    {
        f = vec3(q.z, q.y, q.x);
        faceW = D;
        faceD = W;
    }

    /* f.z = 0 near an outside facade */
    f.z = abs(f.z) - faceD;

    float shelfHeight = 0.285;
    float shelfIndex = floor((f.y + H) / shelfHeight);
    float shelfY = mod(f.y + H, shelfHeight);

    /* Strong outer vertical bookshelf frame */
    float sideFrame = sdBox(
        vec3(
            abs(f.x) - (faceW - 0.018),
            f.y,
            f.z + 0.030
        ),
        vec3(0.020, H, 0.060),
        0.004
    );

    if (sideFrame < d)
    {
        d = sideFrame;
        g_col = vec3(0.045, 0.018, 0.008);
    }

    /* Do shelves only between base and top of the tower */
    if (f.y > -H + 0.02 && f.y < H - 0.02)
    {
        /* Thick horizontal wooden shelf boards */
        float shelfBoard = sdBox(
            vec3(
                f.x,
                shelfY - 0.010,
                f.z + 0.035
            ),
            vec3(faceW - 0.028, 0.012, 0.060),
            0.003
        );

        if (shelfBoard < d)
        {
            d = shelfBoard;
            g_col = vec3(0.10, 0.045, 0.017);
        }

        /* Small warm light strip under each shelf */
        float lamp = sdBox(
            vec3(
                f.x,
                shelfY - (shelfHeight - 0.022),
                f.z + 0.006
            ),
            vec3(faceW * 0.72, 0.006, 0.012),
            0.003
        );

        if (lamp < d)
        {
            d = lamp;
            g_col = vec3(2.2, 1.25, 0.42);
            g_glow = 1.0;
        }

        /* Warm light reaches the books underneath */
        float lampDistance = length(
            vec3(
                f.x * 0.7,
                shelfY - shelfHeight,
                f.z * 0.7
            )
        );

        g_local_lit = exp(-3.2 * lampDistance);

        /* Individual books */
        float bookPitch = 0.052;
        float availableWidth = 2.0 * (faceW - 0.030);

        float bookOffset = f.x + faceW - 0.030;
        float bookIndex = floor(bookOffset / bookPitch);
        float bookX = mod(bookOffset, bookPitch) - bookPitch * 0.5;

        if (bookOffset > 0.0 && bookOffset < availableWidth)
        {
            vec2 bookID = id + vec2(
                shelfIndex * 19.31 + bookIndex * 0.71,
                shelfIndex * 7.17 - bookIndex * 2.13
            );

            float b1 = hash21(bookID);
            float b2 = hash21(bookID + vec2(5.3, 9.1));
            float b3 = hash21(bookID + vec2(11.7, 2.4));
            float b4 = hash21(bookID + vec2(3.6, 15.8));

            /* Leave occasional small gaps */
            if (b1 > 0.055)
            {
                float bookHeight =
                    (shelfHeight - 0.040) *
                    (0.64 + 0.31 * b2);

                float bookDepth =
                    0.045 + 0.026 * b3;

                float bookHalfWidth =
                    bookPitch *
                    (0.40 + 0.08 * b1);

                /*
                   Book rests on the shelf.
                   It protrudes slightly so it is visible
                   over the darker tower wall behind it.
                */
                vec2 bookPos = vec2(
                    bookX,
                    shelfY - (0.020 + bookHeight * 0.5)
                );

                /* A few natural leaning books */
                if (b3 > 0.82)
                {
                    float angle = (b2 - 0.5) * 0.16;
                    float cs = cos(angle);
                    float sn = sin(angle);

                    bookPos = mat2(
                        cs, -sn,
                        sn,  cs
                    ) * bookPos;
                }

                float book = sdBox(
                    vec3(
                        bookPos.x,
                        bookPos.y,
                        f.z + bookDepth * 0.5 - 0.010
                    ),
                    vec3(
                        bookHalfWidth,
                        bookHeight * 0.5,
                        bookDepth * 0.5
                    ),
                    0.003
                );

                if (book < d)
                {
                    d = book;

                    vec3 bcol = bookColour(bookID, b2);

                    /* Gold title bands on selected book spines */
                    if (b4 > 0.68)
                    {
                        float stripe = step(
                            0.90,
                            fract(
                                (bookPos.y + bookHeight * 0.5) *
                                23.0 +
                                b1
                            )
                        );

                        bcol = mix(
                            bcol,
                            vec3(0.95, 0.67, 0.16),
                            stripe * 0.75
                        );
                    }

                    g_col = bcol;
                }
            }
        }
    }

    return d;
}

/* --------------------------------------------------------- */
/* Raymarch helpers                                           */
/* --------------------------------------------------------- */

vec3 normal(vec3 p)
{
    float e = 0.0012;

    return normalize(vec3(
        map(p + vec3(e, 0.0, 0.0)) -
        map(p - vec3(e, 0.0, 0.0)),

        map(p + vec3(0.0, e, 0.0)) -
        map(p - vec3(0.0, e, 0.0)),

        map(p + vec3(0.0, 0.0, e)) -
        map(p - vec3(0.0, 0.0, e))
    ));
}

bool march(inout vec3 p, vec3 rd, out float dd)
{
    dd = 0.0; /* Important: old code did not initialize this */

    for (int i = 0; i < 280; i++)
    {
        float d = map(p);

        if (d < 0.0012)
        {
            return true;
        }

        float stepSize = max(d * 0.65, 0.003);

        p += rd * stepSize;
        dd += stepSize;

        if (dd > 35.0)
        {
            return false;
        }
    }

    return false;
}

float occ(vec3 p, vec3 n, float h)
{
    return clamp(map(p + n * h) / h, 0.0, 1.0);
}

vec3 cameraDir(vec2 uv, vec3 ro, vec3 target, float focalLength)
{
    vec3 forward = normalize(target - ro);
    vec3 right = normalize(cross(vec3(0.0, 1.0, 0.0), forward));
    vec3 up = cross(forward, right);

    return normalize(
        right * uv.x +
        up * uv.y +
        forward * focalLength
    );
}

/* --------------------------------------------------------- */
/* Render                                                     */
/* --------------------------------------------------------- */

vec3 render(vec2 uv)
{
    float lift = wheel.y / MN * 1.1;

    /* Closer, centered camera for clear bookshelf detail */
    vec3 ro = vec3(
        0.0,
        1.55 + lift,
        -9.0
    );

    vec3 target = vec3(
        move.x / MN * 2.0,
        2.45 + lift,
        0.0
    );

    vec3 rd = cameraDir(uv, ro, target, 1.32);

    /* Brighter daylight sky */
    float skyAmount = clamp(rd.y * 0.55 + 0.50, 0.0, 1.0);

    vec3 sky = mix(
        vec3(0.25, 0.32, 0.42),
        vec3(0.77, 0.84, 0.92),
        skyAmount
    );

    vec3 col = sky;

    float dd;
    vec3 p = ro;

    if (march(p, rd, dd))
    {
        vec3 n = normal(p);

        /* Restore material data after normal() calls map() */
        map(p);

        vec3 mat = g_col;
        float glow = g_glow;
        float shelfLight = g_local_lit;

        if (glow > 0.5)
        {
            col = mat * 1.75;
        }
        else
        {
            /* Outdoor sunlight, from top-left/front */
            vec3 sunDir = normalize(vec3(-0.65, 0.85, -0.45));
            vec3 sunCol = vec3(1.20, 1.06, 0.84);

            float sunlight = max(dot(n, sunDir), 0.0);
            float skylight = clamp(n.y * 0.5 + 0.5, 0.0, 1.0);

            /* Small camera fill makes front-facing books clear */
            vec3 cameraLight = normalize(ro - p);
            float frontFill = max(dot(n, cameraLight), 0.0);

            float ao =
                occ(p, n, 0.045) *
                occ(p, n, 0.160);

            vec3 ambient = vec3(0.18, 0.23, 0.31);
            vec3 warmShelf = vec3(1.0, 0.58, 0.22);

            vec3 lighting =
                ambient +
                sunCol * sunlight +
                vec3(0.28, 0.36, 0.50) * skylight +
                vec3(0.45, 0.38, 0.30) * frontFill +
                warmShelf * shelfLight * 0.55;

            col = mat * lighting * ao;

            vec3 halfVec = normalize(sunDir - rd);

            float specular = pow(
                max(dot(n, halfVec), 0.0),
                24.0
            );

            col += sunCol * specular * 0.11;
        }

        /* Softer distance fog, so buildings stay visible */
        float fog = exp(-0.0009 * dd * dd);
        col = mix(sky, col, fog);
    }

    /* Clearer tone mapping than tanh(col * col) */
    col = vec3(1.0) - exp(-col * 1.25);
    col = pow(max(col, vec3(0.0)), vec3(0.4545));

    /* Very subtle vignette: edges do not become black */
    float vignette = clamp(
        1.0 - 0.16 * dot(uv, uv),
        0.78,
        1.0
    );

    col *= vignette;

    return col;
}

void main()
{
    vec2 uv = (FC - 0.5 * R) / MN;

    O = vec4(render(uv), 1.0);
}

#version 300 es
      precision highp float;
      out vec4 O;
      uniform float time;
      uniform vec2 resolution;
      uniform vec2 move;
      uniform vec2 wheel;
      #define FC gl_FragCoord.xy
      #define R resolution
      #define T (25.+time)
      #define S smoothstep
      #define N normalize
      #define MN min(R.x,R.y)
      #define rnd(p) fract(sin(dot(p,vec2(12.9898,78.233)))*345678.)
      #define rot(a) mat2(cos((a)-vec4(0,11,33,0)))

      // Global variables to pass procedural book/shelf properties to the renderer
      vec3 g_col;
      float g_glow;
      float g_local_lit;

      float box(vec3 p, vec3 s, float r) {
      	p=abs(p)-s+r;
      	return length(max(p,.0))+min(.0,max(max(p.x,p.y),p.z))-r;
      }

      float map(vec3 p) {
      	float s=sign(p.y);
      	p.y=abs(p.y)-2.5;
      	vec2 id=floor(p.xz-s);
      	if (mod(id.y,2.)==.0) {
      		p.x-=T*.5;
      		id.x=floor(p.x-s);
      	}
      	p.xz=fract(p.xz-s)-.5;

        // Randomized dimensions for each unique bookshelf building
        float r = rnd(id);
        float W = 0.24 + 0.12 * r;                  // Cabinet Width
        float H = 1.9 - 0.5 * rnd(id + 12.35);      // Cabinet Height
        float D = 0.18;                             // Cabinet Depth

        float d_box = box(p, vec3(W, H, D), 0.015);

        // Set default void/background values
        g_col = vec3(0.01);
        g_glow = 0.0;
        g_local_lit = 0.0;

        // Hierarchical bounding-box check to optimize performance
        if (d_box > 0.15) {
            return d_box;
        }

        float d = d_box;

        // Outer Dark Wood Cabinet Frame
        float d_sides = min(
            box(p - vec3(-W, 0., 0.), vec3(0.012, H, D * 1.02), 0.003),
            box(p - vec3(W, 0., 0.), vec3(0.012, H, D * 1.02), 0.003)
        );
        float d_back = box(p, vec3(W - 0.01, H - 0.01, 0.01), 0.0);
        d = min(d_sides, d_back);

        vec3 c_frame = vec3(0.05, 0.035, 0.018) * (0.6 + 0.4 * r); // Dark rich wood
        g_col = c_frame;

        // Horizontal shelves segmentation
        float sh = 0.35; // Height of each individual shelf
        float y_local = p.y + H;
        float shelf_idx = floor(y_local / sh);
        float y_in_shelf = mod(y_local, sh);

        if (y_local > 0.0 && y_local < 2.0 * H) {
            // Horizontal Shelf Boards
            float d_board = box(p - vec3(0.0, -H + shelf_idx * sh + 0.01, 0.0), vec3(W - 0.01, 0.008, D - 0.01), 0.002);
            if (d_board < d) {
                d = d_board;
                g_col = c_frame * 0.6;
            }

            // Golden lighting strip at the top of each shelf segment
            float d_light = box(vec3(p.x, y_in_shelf - (sh - 0.025), abs(p.z) - (D - 0.04)), vec3(W * 0.85, 0.008, 0.012), 0.002);
            if (d_light < d) {
                d = d_light;
                g_col = vec3(2.5, 1.7, 0.9); // Emissive light strip color
                g_glow = 1.0;
            }

            // Compute local spotlight attenuation for shelf contents
            float dist_to_light = length(vec3(p.x * 0.4, y_in_shelf - sh, abs(p.z) - D));
            g_local_lit = exp(-4.5 * dist_to_light);

            // Individual Books
            float book_w = 0.035;
            float x_local = p.x + W - 0.015;

                vec2 b_id = vec2(shelf_idx, book_idx) + rnd(id) * 45.2;
                float b_rand = rnd(b_id);
                float b_rand2 = rnd(b_id + 7.4);

                float b_h = sh * (0.55 + 0.32 * b_rand);  // Random book height
                float b_d = D * (0.65 + 0.3 * b_rand2);  // Random book depth

                // Rich library color palette
                vec3 b_color = vec3(0.1);
                float col_sel = rnd(b_id + 3.2);
                if (col_sel < 0.22) b_color = vec3(0.38, 0.06, 0.04);       // Deep burgundy red
                else if (col_sel < 0.44) b_color = vec3(0.06, 0.12, 0.26);  // Vintage leather blue
                else if (col_sel < 0.65) b_color = vec3(0.08, 0.22, 0.12);  // Library green
                else if (col_sel < 0.85) b_color = vec3(0.48, 0.32, 0.15);  // Aged brown parchment
                else b_color = vec3(0.55, 0.48, 0.35);                      // Antique paper

                b_color *= (0.65 + 0.35 * rnd(b_id + 9.1));

                // Tilt some books slightly for a natural, dusty library feel
                float tilt = (rnd(b_id + 14.3) - 0.5) * 0.18;
                vec2 rotated_xy = vec2(x_in_book, y_in_shelf - 0.015 - b_h * 0.5);
                if (rnd(b_id + 19.8) > 0.72) {
                    float cos_t = cos(tilt);
                    float sin_t = sin(tilt);
                    rotated_xy = mat2(cos_t, -sin_t, sin_t, cos_t) * rotated_xy;
                }

                // Place the books sitting nicely on the shelves
                vec3 book_pos = vec3(rotated_xy.x, rotated_xy.y, abs(p.z) - (D - b_d * 0.5));
                float d_book = box(book_pos, vec3(book_w * 0.42, b_h * 0.5, b_d * 0.5), 0.003);

                if (d_book < d) {
                    d = d_book;
                    g_col = b_color;

                    // Add golden decorative stripes/gilding to the spines
                    float has_gilding = step(0.68, rnd(b_id + 11.2));
                    if (has_gilding > 0.5 && abs(book_pos.y) < b_h * 0.4) {
                        float bands = step(0.88, cos(book_pos.y * 70.0));
                        g_col = mix(g_col, vec3(0.85, 0.65, 0.2), bands);
                    }
                }
            }
        }

        return d;
      }

      vec3 norm(vec3 p) {
      	float h=1e-3; vec2 k=vec2(-1,1);
      	return N(
      		k.xyy*map(p+k.xyy*h)+
      		k.yxy*map(p+k.yxy*h)+
      		k.yyx*map(p+k.yyx*h)+
      		k.xxx*map(p+k.xxx*h)
      	);
      }

      bool march(inout vec3 p, vec3 rd, out float dd) {
      	for (int i = 0; i < 300; i++) {
      		float d=map(p);
      		if (abs(d)<1e-3) return true;
      		if (dd>15.) return false;
      		p+=rd*d*.5;
      		dd+=d*.5;
      	}
        return false;
      }

      float occ(vec3 p, vec3 n, float d) {
      	return clamp(map(p+n*d)/d,.0,1.);
      }

      vec3 dir(vec2 uv, vec3 p, vec3 t, float z) {
      	vec3 up=vec3(0,1,0),
      	f=N(t-p),
      	r=N(cross(up,f)),
      	u=N(cross(f,r));
      	return mat3(r,u,f)*N(vec3(uv,z));
      }

      void cam(inout vec3 p) {
      	p.xz*=rot(.2-move.x/MN+.2*T*.01);
      }

      vec3 render(vec2 uv) {
      	vec3 col=vec3(0),
      	p=vec3(0,-.3,-23.5-wheel.y/MN-1e2*sin(T*5e-3));
      	cam(p);
      	vec3 rd=dir(uv,p,vec3(0,5.5,0),1.2), lp=p;
      	lp.z+=.5;
      	float dd;
      	if (march(p,rd,dd)) {
      		vec3 n=norm(p);
            
            // Re-evaluate map at the hitpoint to restore procedural color properties
            map(p); 
            
            vec3 l=N(lp-p);
      		float dif=clamp(dot(l,n),.0,1.),
      		spe=pow(clamp(dot(N(lp-rd),n),.0,1.),21.),
      		ao=occ(p,n,.5)*.8*occ(p,n,1.),
      		ld=distance(lp,p), atten=1./(1.+ld*.25+ld*ld*.125);
      		
            vec3 mat=g_col;
            float glow=g_glow;
            float local_lit=g_local_lit;

            // Combine screen-space headlamp with cozy warm shelf light projection
            vec3 warm_lamp = vec3(1.2, 0.85, 0.5) * local_lit * 2.8;
            vec3 diffuse_illum = mat * (dif * atten + warm_lamp);

            col += 0.025 * mat;             // Low ambient base
            col += diffuse_illum * ao;     // Ambient Occlusion + Diffuse Lighting
            col += spe * atten * 0.3;      // Specular glare on cover/woods

            // Ensure our emissive lights glow brightly
            if (glow > 0.5) {
                col = mat;
            }
      	}
      	col=mix(vec3(0),col,exp(-125e-5*dd*dd*dd));
      	col=tanh(col*col);
      	col=sqrt(col);
      	col=mix(vec3(0),col,min(time*.3,1.));
      	// vignette
      	vec2 c=FC/R;
      	c*=1.-c.yx;
      	float vig=c.x*c.y*25.;
      	vig=pow(vig,.5);
      	col*=vig;
      	return col;
      }

      void main() {
      	vec2 uv=(FC-.5*R)/MN;
      	vec3 col=render(uv);
        O=vec4(col,1);
      }

#version 300 es
precision highp float;

out vec4 O;

uniform float time;
uniform vec2 resolution;
uniform vec2 move;

#define FC gl_FragCoord.xy
#define R resolution
#define MN min(R.x, R.y)

/* Utility / Hash */
float hash21(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

float sdBox(vec3 p, vec3 b, float r) {
    vec3 q = abs(p) - b + vec3(r);
    return length(max(q, vec3(0.0))) + min(max(q.x, max(q.y, q.z)), 0.0) - r;
}

vec3 bookColour(vec2 id, float variation) {
    float c = hash21(id + vec2(4.7, 8.3));
    vec3 col;
    if      (c < 0.18) col = vec3(0.68, 0.12, 0.07);   // Rich red
    else if (c < 0.35) col = vec3(0.12, 0.28, 0.16);   // Forest green
    else if (c < 0.50) col = vec3(0.10, 0.20, 0.42);   // Deep blue
    else if (c < 0.70) col = vec3(0.55, 0.32, 0.12);   // Leather brown
    else if (c < 0.84) col = vec3(0.72, 0.52, 0.24);   // Ochre
    else               col = vec3(0.82, 0.76, 0.62);   // Cream
    return col * (0.82 + variation * 0.32);
}

float map(vec3 p) {
    g_col = vec3(0.05, 0.03, 0.015);
    g_glow = 0.0;
    g_local_lit = 0.0;

    /* Repeating monumental wall structure */
    float bayWidth = 3.2;
    float shelfHeight = 0.32;

    float bayX = floor(p.x / bayWidth);
    float localX = mod(p.x, bayWidth) - bayWidth * 0.5;

    float shelfY = mod(p.y, shelfHeight);
    float shelfIndex = floor(p.y / shelfHeight);

    vec2 bayID = vec2(bayX, shelfIndex);

    /* Main bookshelf backing */
    float backWall = abs(p.z - 1.5) - 0.05;
    float d = backWall;
    g_col = vec3(0.05, 0.030, 0.018);

    /* Vertical structural black pillars between bays */
    float pillar = sdBox(
        vec3(localX, p.y, p.z - 0.45),
        vec3(0.08, 100.0, 0.55), 0.01
    );
    if (pillar < d) {
        d = pillar;
        g_col = vec3(0.02, 0.018, 0.016);
    }

    /* Horizontal wooden shelf boards */
    float shelfBoard = sdBox(
        vec3(localX, shelfY - 0.012, p.z - 0.20),
        vec3(bayWidth * 0.48, 0.012, 0.42),
        0.003
    );
    if (shelfBoard < d) {
        d = shelfBoard;
        g_col = vec3(0.16, 0.080, 0.030);
    }

    /* Warm spotlight strip at the top of each shelf section */
    float lamp = sdBox(
        vec3(localX, shelfY - shelfHeight + 0.025, p.z - 0.08),
        vec3(bayWidth * 0.44, 0.006, 0.04),
        0.002
    );
    if (lamp < d) {
        d = lamp;
        g_col = vec3(2.5, 1.4, 0.48);
        g_glow = 1.0;
    }

    /* Warm light reaches the books below */
    float lampDistance = length(vec3(
        localX * 0.5,
        shelfY - shelfHeight,
        (p.z - 0.2) * 0.5
    ));
    g_local_lit = exp(-2.6 * lampDistance);

    /* Individual Books packed tightly on every shelf */
    float bookPitch = 0.052;
    float availableWidth = bayWidth - 0.20;
    float bookOffset = localX + availableWidth * 0.5;
    float bookIndex = floor(bookOffset / bookPitch);
    float bookX = mod(bookOffset, bookPitch) - bookPitch * 0.5;

    if (bookOffset > 0.0 && bookOffset < availableWidth) {
        vec2 bookID = vec2(bayX * 31.7 + bookIndex, shelfIndex * 13.2);

        float b1 = hash21(bookID);
        float b2 = hash21(bookID + vec2(5.3, 9.1));
        float b3 = hash21(bookID + vec2(11.7, 2.4));
        float b4 = hash21(bookID + vec2(3.6, 15.8));

        if (b1 > 0.04) {
            /* Taller books (fills more of the shelf) */
            float bookHeight = (shelfHeight - 0.045) * (0.78 + 0.18 * b2);

            /* Books protrude clearly out from the wall */
            float bookDepth = 0.30 + 0.08 * b3;
            float bookHalfWidth = bookPitch * (0.42 + 0.06 * b1);

            vec2 bookPos = vec2(
                bookX,
                shelfY - (0.024 + bookHeight * 0.5)
            );

            /* Slight natural tilt for realism */
            if (b3 > 0.85) {
                float angle = (b2 - 0.5) * 0.12;
                float cs = cos(angle), sn = sin(angle);
                bookPos = mat2(cs, -sn, sn, cs) * bookPos;
            }

            /* Book spine body, sitting on the shelf */
            float book = sdBox(
                vec3(
                    bookPos.x,
                    bookPos.y,
                    p.z - 0.20 - bookDepth * 0.5 + 0.05
                ),
                vec3(bookHalfWidth, bookHeight * 0.5, bookDepth * 0.5),
                0.003
            );

            if (book < d) {
                d = book;
                vec3 bcol = bookColour(bookID, b2);

                if (b4 > 0.65) {
                    float stripe = step(0.90,
                        fract((bookPos.y + bookHeight * 0.5) * 23.0 + b1));
                    bcol = mix(bcol, vec3(0.95, 0.67, 0.16), stripe * 0.75);
                }
                g_col = bcol;
            }

            /* Visible paper page block on top of books */
            float pageBlock = sdBox(
                vec3(
                    bookPos.x,
                    bookPos.y + bookHeight * 0.5 - 0.003,
                    p.z - 0.20 - bookDepth * 0.5 + 0.05
                ),
                vec3(bookHalfWidth * 0.9, 0.004, bookDepth * 0.75),
                0.001
            );
            if (pageBlock < d) {
                d = pageBlock;
                g_col = vec3(0.88, 0.82, 0.70);
            }
        }
    }

    return d;
}

vec3 normal(vec3 p) {
    float e = 0.0012;
    return normalize(vec3(
        map(p + vec3(e, 0.0, 0.0)) - map(p - vec3(e, 0.0, 0.0)),
        map(p + vec3(0.0, e, 0.0)) - map(p - vec3(0.0, e, 0.0)),
        map(p + vec3(0.0, 0.0, e)) - map(p - vec3(0.0, 0.0, e))
    ));
}

bool march(inout vec3 p, vec3 rd, out float dd) {
    dd = 0.0;
    for (int i = 0; i < 320; i++) {
        float d = map(p);
        if (abs(d) < 0.001) return true;
        float stepSize = max(d * 0.55, 0.002);
        p += rd * stepSize;
        dd += stepSize;
        if (dd > 50.0) break;
    }
    return false;
}

float occ(vec3 p, vec3 n, float h) {
    return clamp(map(p + n * h) / h, 0.0, 1.0);
}

vec3 cameraDir(vec2 uv, vec3 ro, vec3 target, float focalLength) {
    vec3 forward = normalize(target - ro);
    vec3 right = normalize(cross(vec3(0.0, 1.0, 0.0), forward));
    vec3 up = cross(forward, right);
    return normalize(right * uv.x + up * uv.y + forward * focalLength);
}

vec3 render(vec2 uv) {
    /* Smooth camera panning along the grand library wall */
    float panX = sin(time * 0.1) * 2.5 + move.x / MN * 2.0;
    float lift = 2.2 + wheel.y / MN * 1.5 + sin(time * 0.05) * 0.5;

    vec3 ro = vec3(panX, lift, -4.5);
    vec3 target = vec3(panX, lift + 0.2, 2.0);
    vec3 rd = cameraDir(uv, ro, target, 1.30);

    /* Warm dark hall background (slightly brighter than before) */
    vec3 sky = vec3(0.10, 0.08, 0.06);
    vec3 col = sky;

    float dd;
    vec3 p = ro;

    if (march(p, rd, dd)) {
        vec3 n = normal(p);
        map(p);

        vec3 mat = g_col;
        float glow = g_glow;
        float shelfLight = g_local_lit;

        if (glow > 0.5) {
            col = mat * 2.4;
        } else {
            vec3 sunDir = normalize(vec3(-0.3, 1.0, -0.5));
            vec3 sunCol = vec3(1.35, 1.20, 0.95);

            float sunlight = max(dot(n, sunDir), 0.0);
            float skylight = clamp(n.y * 0.5 + 0.5, 0.0, 1.0);
            vec3 cameraLight = normalize(ro - p);
            float frontFill = max(dot(n, cameraLight), 0.0);

            float ao = occ(p, n, 0.04) * occ(p, n, 0.15);

            /* Brighter ambient + stronger warm shelf light */
            vec3 ambient = vec3(0.22, 0.20, 0.16);
            vec3 warmShelf = vec3(1.30, 0.78, 0.30);

            vec3 lighting =
                ambient +
                sunCol * sunlight * 0.85 +
                vec3(0.30, 0.30, 0.34) * skylight +
                vec3(0.40, 0.35, 0.30) * frontFill +
                warmShelf * shelfLight * 1.20;

            col = mat * lighting * ao;

            vec3 halfVec = normalize(sunDir - rd);
            float specular = pow(max(dot(n, halfVec), 0.0), 32.0);
            col += sunCol * specular * 0.18;
        }

        /* Softer fog so distant shelves remain visible */
        float fog = exp(-0.006 * dd * dd);
        col = mix(sky, col, fog);
    }

    /* Cleaner tone mapping */
    col = vec3(1.0) - exp(-col * 1.15);
    col = pow(max(col, vec3(0.0)), vec3(0.4545));

    /* Subtle vignette */
    float vignette = clamp(1.0 - 0.12 * dot(uv, uv), 0.85, 1.0);
    col *= vignette;

    return col;
}

void main() {
    vec2 uv = (FC - 0.5 * R) / MN;
    O = vec4(render(uv), 1.0);
}
