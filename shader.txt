Different versions 



Perfect — keep your current file as **default = bookshelves**.

Below are **shaders only**. All use the same uniforms `time, resolution, move, wheel` so your pause button, mouse + wheel, and resolution toggle keep working.

**How to use:** click ✏️, Ctrl+A, paste one shader, wait 1s. Click ⏮️ to go back to default bookshelves.

### 1. Neon Night Library - cyberpunk books

```glsl
#version 300 es
precision highp float;
out vec4 O;
uniform float time;
uniform vec2 resolution;
uniform vec2 move;
uniform vec2 wheel;
#define FC gl_FragCoord.xy
#define R resolution
#define T (25.0+time)
#define MN min(R.x,R.y)
#define rnd(p) fract(sin(dot(p,vec2(12.9898,78.233)))*345678.0)
#define rot(a) mat2(cos((a)-vec4(0,11,33,0)))
const float SHELF_H=0.180;
const float BOOK_W=0.024;
const float POST_W=0.024;
float sdBox(vec3 p,vec3 b,float r){vec3 q=abs(p)-b;return length(max(q,vec3(0.0)))+min(max(q.x,max(q.y,q.z)),0.0)-r;}
struct Tower{vec3 p;vec2 id;float W,H,D,sy;};
Tower getTower(vec3 p){Tower t;float sy=p.y<0.0?-1.0:1.0;p.y=abs(p.y)-2.5;vec2 id=floor(p.xz-sy);if(mod(id.y,2.0)==0.0){p.x-=T*0.5;id.x=floor(p.x-sy);}p.xz=fract(p.xz-sy)-0.5;float r1=rnd(id);float r2=rnd(id+7.31);t.p=p;t.id=id;t.sy=sy;t.W=0.150+0.160*r1;t.D=0.140+0.120*r2;t.H=max(1.35,2.05-0.55*rnd(id+12.35));return t;}
struct Facade{vec3 col;float inset;float glow;float lit;};
vec3 bookPalette(vec2 id){float s=rnd(id+2.11);if(s<0.20)return vec3(0.10,0.85,1.0);else if(s<0.38)return vec3(1.0,0.20,0.75);else if(s<0.55)return vec3(0.35,0.30,1.0);else if(s<0.70)return vec3(0.15,1.0,0.55);else if(s<0.85)return vec3(1.0,0.55,0.15);else return vec3(0.90,0.95,1.0);}
Facade facade(vec2 uv,float halfW,float H,vec2 id,float faceSeed,float worldDist){
Facade f;float r1=rnd(id);vec3 frameCol=mix(vec3(0.015,0.020,0.045),vec3(0.040,0.060,0.110),r1);
f.col=frameCol;f.inset=0.0;f.glow=0.0;f.lit=0.0;
if(abs(uv.x)>halfW-POST_W){f.col=frameCol*0.50;return f;}
if(uv.y>H-0.035){f.col=frameCol*0.80;return f;}
if(uv.y<-H+0.040){f.col=frameCol*0.55;return f;}
float yLocal=uv.y+H;float shelfIdx=floor(yLocal/SHELF_H);float yInShelf=mod(yLocal,SHELF_H);float boardT=0.017;
if(yInShelf<boardT){f.col=frameCol*0.90;f.inset=0.001;return f;}
if(yInShelf>SHELF_H-0.012){f.col=vec3(0.35,2.0,2.6);f.glow=1.0;f.inset=0.003;return f;}
f.lit=exp(-5.5*(SHELF_H-yInShelf));
float lod=smoothstep(8.0,18.0,worldDist);
if(lod>0.85){f.col=mix(frameCol,vec3(0.10,0.35,0.50),fract(shelfIdx*0.3));f.inset=0.015;return f;}
float off=uv.x-(-halfW+POST_W);
if(off<0.0||off>(halfW-POST_W)*2.0){f.col=frameCol*0.40;f.inset=0.020;return f;}
float bookIdx=floor(off/BOOK_W);float xInBook=mod(off,BOOK_W)-BOOK_W*0.5;
vec2 bId=vec2(shelfIdx*39.17+bookIdx+faceSeed*97.3,id.x*21.71+id.y*5.91-bookIdx*1.37);
float b1=rnd(bId),b2=rnd(bId+5.77),b3=rnd(bId+11.33),b4=rnd(bId+19.91),b5=rnd(bId+31.44);
if(b1<0.055){f.col=vec3(0.005,0.010,0.020);f.inset=0.050;return f;}
float bookBase=boardT+0.002;float bookH=(SHELF_H-bookBase-0.014)*(0.58+0.38*b2);float bookHalfW=BOOK_W*(0.34+0.14*b1);
vec2 bp=vec2(xInBook,yInShelf-(bookBase+bookH*0.5));
if(b3>0.86){float a=(b2-0.5)*0.12;float cs=cos(a),sn=sin(a);bp=mat2(cs,-sn,sn,cs)*bp;}
if(!(abs(bp.x)<bookHalfW&&abs(bp.y)<bookH*0.5)){f.col=vec3(0.008,0.012,0.022);f.inset=0.045;return f;}
vec3 bcol=bookPalette(bId)*(0.90+0.40*b5);
float edge=smoothstep(0.55,0.98,abs(bp.x)/max(bookHalfW,0.0001));bcol*=1.0-edge*0.30;
float contact=1.0-smoothstep(0.0,0.014,bp.y+bookH*0.5);bcol*=1.0-contact*0.35;
if(b4>0.50){float bnd1=1.0-smoothstep(0.001,0.005,abs(bp.y-bookH*0.25));float bnd2=1.0-smoothstep(0.001,0.005,abs(bp.y+bookH*0.25));bcol=mix(bcol,vec3(0.80,1.0,1.0),max(bnd1,bnd2)*0.85);}
if(bp.y>bookH*0.5-0.006)bcol=mix(bcol,vec3(0.75,0.90,1.0),0.85);
bcol+=vec3(0.05,0.25,0.35)*f.lit;
f.col=bcol;f.inset=0.004+0.010*b3;return f;}
Facade towerFacade(Tower t,float wd){vec3 p=t.p;float uy=p.y*t.sy;if(abs(p.z)/t.D>=abs(p.x)/t.W)return facade(vec2(p.x,uy),t.W,t.H,t.id,sign(p.z),wd);else return facade(vec2(p.z,uy),t.D,t.H,t.id,2.0+sign(p.x),wd);}
float map(vec3 p){Tower t=getTower(p);float d=sdBox(t.p,vec3(t.W,t.H,t.D),0.006);if(d<0.080)d+=towerFacade(t,length(p)).inset;return d;}
vec3 calcNormal(vec3 p){float e=0.0012;vec2 k=vec2(-1.0,1.0);return normalize(k.xyy*map(p+k.xyy*e)+k.yxy*map(p+k.yxy*e)+k.yyx*map(p+k.yyx*e)+k.xxx*map(p+k.xxx*e));}
bool march(inout vec3 p,vec3 rd,out float dd){dd=0.0;for(int i=0;i<430;i++){float d=map(p);if(d<0.0010)return true;if(dd>26.0)return false;float st=max(d*0.40,0.0010);p+=rd*st;dd+=st;}return false;}
float calcAO(vec3 p,vec3 n){float o=0.0,s=1.0;for(int i=0;i<4;i++){float h=0.010+0.045*float(i);o+=(h-map(p+n*h))*s;s*=0.60;}return clamp(1.0-1.8*o,0.14,1.0);}
vec3 camDir(vec2 uv,vec3 ro,vec3 ta,float fl){vec3 f=normalize(ta-ro);vec3 r=normalize(cross(vec3(0.0,1.0,0.0),f));return normalize(r*uv.x+cross(f,r)*uv.y+f*fl);}
vec3 render(vec2 uv){vec3 ro=vec3(0.0,-0.3,-23.5-wheel.y/MN-100.0*sin(T*0.005));ro.xz=rot(0.20-move.x/MN+0.20*T*0.01)*ro.xz;vec3 rd=camDir(uv,ro,vec3(0.0,5.5,0.0),1.2);vec3 sky=vec3(0.0);vec3 col=sky;float dd=0.0;vec3 p=ro;
if(march(p,rd,dd)){vec3 n=calcNormal(p);Facade f=towerFacade(getTower(p),dd);
if(f.glow>0.5)col=f.col*1.7;else{vec3 sd=normalize(vec3(-0.55,0.72,-0.40));vec3 sc=vec3(0.70,0.90,1.10);float dif=max(dot(n,sd),0.0);float skyl=clamp(n.y*0.5+0.5,0.0,1.0);float fill=max(dot(n,normalize(ro-p)),0.0);float ao=calcAO(p,n);
vec3 li=vec3(0.22,0.26,0.38)+sc*dif*1.10+vec3(0.15,0.30,0.50)*skyl*0.50+vec3(0.25,0.35,0.50)*fill*0.60+vec3(0.30,1.20,1.60)*f.lit*1.10;col=f.col*li*ao;
col+=sc*pow(max(dot(n,normalize(sd-rd)),0.0),28.0)*0.18*ao;col+=vec3(0.20,0.50,0.80)*pow(1.0-max(dot(n,-rd),0.0),4.0)*0.35;}}
col=mix(sky,col,exp(-125e-5*dd*dd*dd));col=vec3(1.0)-exp(-col*1.70);col=pow(max(col,vec3(0.0)),vec3(0.4545));col=mix(vec3(0.0),col,smoothstep(0.0,0.3,time));vec2 c=FC/R;c*=1.0-c.yx;col*=pow(clamp(c.x*c.y*25.0,0.0,1.0),0.14);return col;}
void main(){vec2 uv=(FC-0.5*R)/MN;O=vec4(render(uv),1.0);}
```

### 3. Original Copper City - fixed, no uninitialized dd

```glsl
#version 300 es
precision highp float;
out vec4 O;
uniform float time;
uniform vec2 resolution;
uniform vec2 move;
uniform vec2 wheel;
#define FC gl_FragCoord.xy
#define R resolution
#define T (25.+time)
#define N normalize
#define MN min(R.x,R.y)
#define rnd(p) fract(sin(dot(p,vec2(12.9898,78.233)))*345678.)
#define rot(a) mat2(cos((a)-vec4(0,11,33,0)))
float box(vec3 p,vec3 s,float r){p=abs(p)-s+r;return length(max(p,0.0))+min(0.0,max(max(p.x,p.y),p.z))-r;}
float map(vec3 p){float s=sign(p.y);p.y=abs(p.y)-2.5;vec2 id=floor(p.xz-s);if(mod(id.y,2.0)==0.0){p.x-=T*0.5;id.x=floor(p.x-s);}float f=1.0-dot(abs(fract(p*42.0)-0.5)-0.25,vec3(1.0))*0.5;p.xz=fract(p.xz-s)-0.5;return box(p,vec3(0.10+0.30*rnd(id),2.0-0.60*rnd(id),0.20),f*f*0.0125)-1e-3*f;}
vec3 norm(vec3 p){float h=1e-3;vec2 k=vec2(-1.0,1.0);return N(k.xyy*map(p+k.xyy*h)+k.yxy*map(p+k.yxy*h)+k.yyx*map(p+k.yyx*h)+k.xxx*map(p+k.xxx*h));}
bool march(inout vec3 p,vec3 rd,out float dd){dd=0.0;for(int i=0;i<400;i++){float d=map(p);if(abs(d)<1e-3)return true;if(dd>15.0)return false;p+=rd*d*0.5;dd+=d*0.5;}return false;}
float occ(vec3 p,vec3 n,float d){return clamp(map(p+n*d)/d,0.0,1.0);}
vec3 dir(vec2 uv,vec3 p,vec3 t,float z){vec3 up=vec3(0.0,1.0,0.0);vec3 f=N(t-p);vec3 r=N(cross(up,f));vec3 u=N(cross(f,r));return mat3(r,u,f)*N(vec3(uv,z));}
void cam(inout vec3 p){p.xz*=rot(0.2-move.x/MN+0.2*T*0.01);}
vec3 render(vec2 uv){vec3 col=vec3(0.0);vec3 p=vec3(0.0,-0.3,-23.5-wheel.y/MN-1e2*sin(T*5e-3));cam(p);vec3 rd=dir(uv,p,vec3(0.0,5.5,0.0),1.2);vec3 lp=p;lp.z+=0.5;float dd=0.0;
if(march(p,rd,dd)){vec3 n=norm(p);vec3 l=N(lp-p);float dif=clamp(dot(l,n),0.0,1.0);float spe=pow(clamp(dot(N(lp-rd),n),0.0,1.0),21.0);float ao=occ(p,n,0.5)*0.8*occ(p,n,1.0);float ld=distance(lp,p);float atten=1.0/(1.0+ld*0.25+ld*ld*0.125);vec3 mat=vec3(4.0,1.6,0.6);col+=0.08+dif*mat*ao*atten;col+=spe*atten;}
col=mix(vec3(0.0),col,exp(-125e-5*dd*dd*dd));col=tanh(col*col);col=sqrt(col);col=mix(vec3(0.0),col,min(time*0.3,1.0));vec2 c=FC/R;c*=1.0-c.yx;float vig=c.x*c.y*25.0;vig=pow(vig,0.5);col*=vig;return col;}
void main(){vec2 uv=(FC-0.5*R)/MN;O=vec4(render(uv),1.0);}
```

### 4. Warp Tunnel - light abstract, mouse rotates, wheel speeds

```glsl
#version 300 es
precision highp float;
out vec4 O;
uniform float time;
uniform vec2 resolution;
uniform vec2 move;
uniform vec2 wheel;
#define FC gl_FragCoord.xy
#define R resolution
#define MN min(R.x,R.y)
float hash21(vec2 p){return fract(sin(dot(p,vec2(127.1,311.7)))*43758.5453;}
void main(){
vec2 uv=(FC-0.5*R)/MN;
float t=time*(1.0+clamp(wheel.y/MN*0.3,-0.5,2.0));
float r0=move.x/MN*0.8;float ca=cos(r0),sa=sin(r0);uv=mat2(ca,-sa,sa,ca)*uv;
float r=length(uv);float a=atan(uv.y,uv.x);
float z=1.0/(r+0.15);
vec2 tuv=vec2(a*3.0,z-t*1.5);
vec2 gid=floor(tuv*vec2(8.0,4.0));
float star=step(0.92,hash21(gid));
float tw=0.5+0.5*sin(time*8.0+hash21(gid)*40.0);
vec3 col=vec3(0.0);
col+=vec3(0.15,0.05,0.30)*exp(-r*2.0);
col+=vec3(0.0,0.20,0.35)*exp(-abs(r-0.4)*4.0);
col+=vec3(1.0,0.6,0.2)*star*tw*smoothstep(1.5,0.2,r);
col+=vec3(0.8,0.9,1.0)*exp(-r*8.0)*1.5;
float lines=fract(a*10.0+z*2.0-t*3.0);
col+=vec3(0.2,0.5,0.8)*smoothstep(0.90,1.0,lines)*exp(-r*1.5)*0.5;
col=vec3(1.0)-exp(-col*1.6);
col=pow(max(col,vec3(0.0)),vec3(0.4545));
vec2 c=FC/R;c*=1.0-c.yx;
col*=pow(clamp(c.x*c.y*25.0,0.0,1.0),0.20);
O=vec4(col,1.0);
}
```

## 1. Neon Mirrored Towers

An infinite mirrored city with cyan/magenta windows and illuminated building edges. Uses ray–box intersections rather than displaced surfaces.

```glsl
#version 300 es
precision highp float;

out vec4 O;

uniform float time;
uniform vec2 resolution;
uniform vec2 move;
uniform vec2 wheel;

const float CELL = 1.20;
const float FAR_DISTANCE = 36.0;

float hash21(vec2 p)
{
    vec3 q = fract(vec3(p.x, p.y, p.x) * 0.1031);
    q += dot(q, q.yzx + 33.33);
    return fract((q.x + q.y) * q.z);
}

void main()
{
    float mn = max(min(resolution.x, resolution.y), 1.0);
    vec2 uv = (gl_FragCoord.xy - 0.5 * resolution) / mn;

    float yaw = 0.24 - move.x / mn * 0.55;
    float pitch = clamp(
        0.06 + move.y / mn * 0.25,
        -0.35,
        0.35
    );

    vec3 forward = normalize(vec3(
        sin(yaw),
        tan(pitch),
        cos(yaw)
    ));

    vec3 right = normalize(
        cross(vec3(0.0, 1.0, 0.0), forward)
    );
    vec3 up = cross(forward, right);

    vec3 ro = vec3(
        0.05,
        -0.10,
        -9.0 + time * 0.25 + wheel.y / mn * 2.0
    );

    vec3 rd = normalize(
        forward * 1.30 +
        right * uv.x +
        up * uv.y
    );

    // Safe reciprocal direction for ray-box intersections.
    vec3 raySign = mix(
        vec3(-1.0),
        vec3(1.0),
        step(vec3(0.0), rd)
    );

    vec3 invDirection =
        raySign / max(abs(rd), vec3(0.000001));

    // Traverse the infinite XZ grid in ray order.
    vec2 cell = floor(ro.xz / CELL);

    vec2 nextBoundary =
        (
            (cell + step(vec2(0.0), rd.xz)) * CELL -
            ro.xz
        ) * invDirection.xz;

    vec2 boundaryStep = CELL * abs(invDirection.xz);

    float entered = 0.0;
    float hitDistance = FAR_DISTANCE;

    vec3 hitLocal = vec3(0.0);
    vec3 hitSize = vec3(1.0);
    vec2 hitID = vec2(0.0);

    bool hit = false;

    for (int i = 0; i < 96; i++)
    {
        if (entered > FAR_DISTANCE)
        {
            break;
        }

        vec3 halfSize = vec3(
            0.20 + 0.15 * hash21(cell),
            1.45 + 0.55 * hash21(cell + vec2(4.0, 17.0)),
            0.18 + 0.13 * hash21(cell + vec2(9.0, 3.0))
        );

        // One tower above and one below the central passage.
        for (int sideIndex = 0; sideIndex < 2; sideIndex++)
        {
            float side = sideIndex == 0 ? -1.0 : 1.0;

            vec3 center = vec3(
                (cell.x + 0.5) * CELL,
                side * 2.65,
                (cell.y + 0.5) * CELL
            );

            vec3 a =
                (center - halfSize - ro) * invDirection;

            vec3 b =
                (center + halfSize - ro) * invDirection;

            vec3 nearPlane = min(a, b);
            vec3 farPlane = max(a, b);

            float nearT = max(
                nearPlane.x,
                max(nearPlane.y, nearPlane.z)
            );

            float farT = min(
                farPlane.x,
                min(farPlane.y, farPlane.z)
            );

            if (
                nearT >= 0.0 &&
                nearT <= farT &&
                nearT < hitDistance
            )
            {
                hit = true;
                hitDistance = nearT;
                hitLocal = ro + rd * nearT - center;
                hitSize = halfSize;
                hitID = cell;
            }
        }

        if (hit)
        {
            break;
        }

        if (nextBoundary.x < nextBoundary.y)
        {
            entered = nextBoundary.x;
            nextBoundary.x += boundaryStep.x;
            cell.x += raySign.x;
        }
        else
        {
            entered = nextBoundary.y;
            nextBoundary.y += boundaryStep.y;
            cell.y += raySign.z;
        }
    }

    vec3 color = vec3(0.0);

    if (hit)
    {
        vec3 relativeFace = abs(hitLocal) / hitSize;
        vec3 n;

        if (
            relativeFace.y > relativeFace.x &&
            relativeFace.y > relativeFace.z
        )
        {
            n = vec3(0.0, sign(hitLocal.y), 0.0);
        }
        else if (relativeFace.x > relativeFace.z)
        {
            n = vec3(sign(hitLocal.x), 0.0, 0.0);
        }
        else
        {
            n = vec3(0.0, 0.0, sign(hitLocal.z));
        }

        vec3 neon = mix(
            vec3(0.02, 0.72, 1.00),
            vec3(1.00, 0.045, 0.46),
            hash21(hitID + vec2(30.0, 8.0))
        );

        vec3 lightDirection =
            normalize(vec3(-0.6, 0.9, -0.4));

        float diffuse = max(dot(n, lightDirection), 0.0);

        color = vec3(0.025, 0.032, 0.050) *
                (0.35 + diffuse * 1.10);

        float footprint = clamp(
            hitDistance /
            (
                mn * 1.30 *
                max(dot(n, -rd), 0.12)
            ),
            0.0001,
            0.25
        );

        if (abs(n.y) < 0.5)
        {
            float horizontal = abs(n.x) > 0.5
                ? hitLocal.z
                : hitLocal.x;

            float halfWidth = abs(n.x) > 0.5
                ? hitSize.z
                : hitSize.x;

            vec2 grid = vec2(
                horizontal + halfWidth,
                hitLocal.y + hitSize.y
            ) / vec2(0.085, 0.150);

            vec2 localWindow = abs(fract(grid) - 0.5);
            vec2 aa = footprint / vec2(0.085, 0.150);

            vec2 windowMask = vec2(1.0) - smoothstep(
                vec2(0.28, 0.32) - aa,
                vec2(0.28, 0.32) + aa,
                localWindow
            );

            float occupied = step(
                0.27,
                hash21(floor(grid) + hitID * 17.0)
            );

            float windows =
                windowMask.x * windowMask.y * occupied;

            // Blend subpixel windows toward their average coverage.
            windows = mix(
                windows,
                0.26,
                smoothstep(0.012, 0.065, footprint)
            );

            float border = min(
                halfWidth - abs(horizontal),
                hitSize.y - abs(hitLocal.y)
            );

            float edge = 1.0 - smoothstep(
                0.004,
                0.004 + footprint * 1.4,
                border
            );

            color += neon * (windows * 2.0 + edge * 0.85);
        }
        else
        {
            float border = min(
                hitSize.x - abs(hitLocal.x),
                hitSize.z - abs(hitLocal.z)
            );

            float edge = 1.0 - smoothstep(
                0.004,
                0.004 + footprint,
                border
            );

            color += neon * edge * 0.85;
        }

        color *= exp(-0.004 * hitDistance * hitDistance);
    }

    color = vec3(1.0) - exp(-color);
    color = pow(max(color, vec3(0.0)), vec3(0.4545));

    O = vec4(color, 1.0);
}
```

## 2. Neon Warp Tunnel

A continuous glowing tunnel with filtered rings and purple structural lines.

```glsl
#version 300 es
precision highp float;

out vec4 O;

uniform float time;
uniform vec2 resolution;
uniform vec2 move;
uniform vec2 wheel;

const float TAU = 6.28318530718;

// Box-filtered periodic stripe.
// Filtering happens before the repeating pattern is sampled.
float filteredStripe(float coordinate, float duty, float width)
{
    width = max(width, 0.0001);

    float a = coordinate - width * 0.5;
    float b = coordinate + width * 0.5;

    float coverage =
        (floor(b) - floor(a)) * duty +
        min(fract(b), duty) -
        min(fract(a), duty);

    return clamp(coverage / width, 0.0, 1.0);
}

void main()
{
    float mn = max(min(resolution.x, resolution.y), 1.0);
    vec2 uv = (gl_FragCoord.xy - 0.5 * resolution) / mn;

    uv += clamp(
        move / mn,
        vec2(-2.0),
        vec2(2.0)
    ) * 0.15;

    vec3 rd = normalize(vec3(uv, 1.20));

    // Analytical intersection with the inside of a cylinder.
    float radius = 2.40;

    float distanceToWall = min(
        radius / max(length(rd.xy), 0.000001),
        80.0
    );

    vec3 p = rd * distanceToWall;

    float angle = atan(p.y, p.x + 0.000001);
    float angularCoordinate = angle / TAU * 16.0;

    float lengthCoordinate =
        (
            p.z +
            time * 2.80 +
            wheel.y / mn * 4.0
        ) / 1.35;

    // Angular footprint avoids differentiating across atan's seam.
    float angularWidth = max(
        length(fwidth(p.xy)) /
        max(length(p.xy), 0.001) *
        16.0 / TAU,
        0.0001
    );

    float lengthWidth = max(
        fwidth(lengthCoordinate),
        0.0001
    );

    float rings = filteredStripe(
        lengthCoordinate + 0.025,
        0.050,
        lengthWidth
    );

    float ribs = filteredStripe(
        angularCoordinate + 0.014,
        0.028,
        angularWidth
    );

    float ringDistance =
        abs(fract(lengthCoordinate + 0.5) - 0.5);

    float ringGlow = exp(-18.0 * ringDistance);

    ringGlow = mix(
        ringGlow,
        0.111,
        smoothstep(0.05, 0.80, lengthWidth)
    );

    float ribDistance =
        abs(fract(angularCoordinate + 0.5) - 0.5);

    float ribGlow = exp(-22.0 * ribDistance);

    ribGlow = mix(
        ribGlow,
        0.091,
        smoothstep(0.05, 0.80, angularWidth)
    );

    vec3 ringColor = mix(
        vec3(0.015, 0.85, 1.00),
        vec3(1.00, 0.03, 0.40),
        0.5 + 0.5 * sin(
            lengthCoordinate * 0.35 - time * 0.20
        )
    );

    vec3 ribColor = vec3(0.40, 0.045, 1.00);

    vec3 color = vec3(0.006, 0.009, 0.020);

    color += ringColor * (rings * 2.80 + ringGlow * 0.22);
    color += ribColor * (ribs * 2.00 + ribGlow * 0.15);

    float fade =
        exp(-0.045 * distanceToWall) *
        (1.0 - smoothstep(55.0, 80.0, distanceToWall));

    color *= fade;

    color = vec3(1.0) - exp(-color * 1.25);
    color = pow(max(color, vec3(0.0)), vec3(0.4545));

    O = vec4(color, 1.0);
}
```

## 3. Liquid Chrome

A slowly morphing metallic sculpture with procedural studio-light reflections. Drag to orbit; scroll to change viewing distance.

```glsl
#version 300 es
precision highp float;

out vec4 O;

uniform float time;
uniform vec2 resolution;
uniform vec2 move;
uniform vec2 wheel;

const float TAU = 6.28318530718;

float smoothUnion(float a, float b, float radius)
{
    float h = clamp(
        0.5 + 0.5 * (b - a) / radius,
        0.0,
        1.0
    );

    return mix(b, a, h) - radius * h * (1.0 - h);
}

float scene(vec3 p)
{
    float d = length(p) - 0.72;

    for (int i = 0; i < 5; i++)
    {
        float index = float(i);
        float angle = index * TAU / 5.0 + time * 0.30;

        vec3 center = vec3(
            cos(angle) * 0.83,
            sin(angle * 1.7 + time * 0.25) * 0.43,
            sin(angle) * 0.83
        );

        float radius =
            0.40 + 0.055 * sin(time * 0.70 + index);

        d = smoothUnion(
            d,
            length(p - center) - radius,
            0.28
        );
    }

    return d;
}

vec3 normalAt(vec3 p)
{
    vec2 e = vec2(0.001, 0.0);

    return normalize(vec3(
        scene(p + e.xyy) - scene(p - e.xyy),
        scene(p + e.yxy) - scene(p - e.yxy),
        scene(p + e.yyx) - scene(p - e.yyx)
    ));
}

// Procedural environment reflected by the metal.
vec3 environment(vec3 direction)
{
    float sky = smoothstep(-0.6, 0.8, direction.y);

    vec3 color = mix(
        vec3(0.012, 0.016, 0.025),
        vec3(0.080, 0.110, 0.160),
        sky
    );

    vec3 keyDirection =
        normalize(vec3(-0.45, 0.75, -0.60));

    float key = pow(
        max(dot(direction, keyDirection), 0.0),
        48.0
    );

    color += vec3(4.0, 3.8, 3.4) * key;

    float strip = exp(
        -pow((direction.x - 0.25) * 14.0, 2.0)
    );

    strip *= smoothstep(-0.30, 0.40, direction.y);

    color += vec3(1.5, 1.8, 2.3) * strip;

    float cyan = pow(
        max(
            dot(direction, normalize(vec3(0.8, 0.2, -0.5))),
            0.0
        ),
        14.0
    );

    float orange = pow(
        max(
            dot(direction, normalize(vec3(-0.7, 0.1, -0.8))),
            0.0
        ),
        18.0
    );

    color += vec3(0.08, 0.90, 1.80) * cyan;
    color += vec3(1.80, 0.48, 0.12) * orange;

    return color;
}

vec3 renderSample(vec2 uv, float mn)
{
    float yaw = 0.60 + time * 0.12 - move.x / mn;

    float pitch = clamp(
        0.15 + move.y / mn * 0.6,
        -0.80,
        0.80
    );

    float cameraDistance = clamp(
        4.50 + wheel.y / mn * 1.20,
        2.80,
        7.50
    );

    vec3 ro = cameraDistance * vec3(
        sin(yaw) * cos(pitch),
        sin(pitch),
        -cos(yaw) * cos(pitch)
    );

    vec3 forward = normalize(-ro);
    vec3 right = normalize(
        cross(vec3(0.0, 1.0, 0.0), forward)
    );
    vec3 up = cross(forward, right);

    vec3 rd = normalize(
        forward * 1.40 +
        right * uv.x +
        up * uv.y
    );

    // Start tracing at a bounding sphere around the sculpture.
    float b = dot(ro, rd);
    float discriminant = b * b - dot(ro, ro) + 4.0;

    if (discriminant < 0.0)
    {
        return vec3(0.0);
    }

    float root = sqrt(discriminant);
    float t = max(0.0, -b - root);
    float endT = -b + root;

    bool hit = false;
    vec3 p = ro;

    for (int i = 0; i < 112; i++)
    {
        if (t > endT)
        {
            break;
        }

        p = ro + rd * t;
        float d = scene(p);

        if (d < 0.0007)
        {
            hit = true;
            break;
        }

        t += d * 0.85;
    }

    if (!hit)
    {
        return vec3(0.0);
    }

    vec3 n = normalAt(p);
    vec3 reflected = reflect(rd, n);

    float facing = clamp(dot(n, -rd), 0.0, 1.0);
    float fresnel = pow(1.0 - facing, 5.0);

    vec3 color = environment(reflected);
    color *= vec3(0.90, 0.96, 1.0);
    color *= 0.72 + 0.28 * fresnel;

    float fill = max(
        dot(n, normalize(vec3(-0.4, 0.9, -0.6))),
        0.0
    );

    color += vec3(0.035) * fill;

    return color;
}

void main()
{
    float mn = max(min(resolution.x, resolution.y), 1.0);
    vec2 uv = (gl_FragCoord.xy - 0.5 * resolution) / mn;

    // Two spatial samples soften silhouette aliasing.
    vec2 offset = vec2(0.25) / mn;

    vec3 color = 0.5 * (
        renderSample(uv - offset, mn) +
        renderSample(uv + offset, mn)
    );

    color = vec3(1.0) - exp(-color * 1.25);
    color = pow(max(color, vec3(0.0)), vec3(0.4545));

    O = vec4(color, 1.0);
}
```

## 4. Cosmic Nebula

Slow-moving blue and magenta clouds with layered, gently twinkling stars. No textures are required.

```glsl
#version 300 es
precision highp float;

out vec4 O;

uniform float time;
uniform vec2 resolution;
uniform vec2 move;
uniform vec2 wheel;

const float TAU = 6.28318530718;

float hash21(vec2 p)
{
    vec3 q = fract(vec3(p.x, p.y, p.x) * 0.1031);
    q += dot(q, q.yzx + 33.33);
    return fract((q.x + q.y) * q.z);
}

vec2 hash22(vec2 p)
{
    return vec2(
        hash21(p),
        hash21(p + vec2(17.13, 91.71))
    );
}

float noise2(vec2 p)
{
    vec2 cell = floor(p);
    vec2 local = fract(p);

    vec2 u = local * local * (3.0 - 2.0 * local);

    float a = hash21(cell);
    float b = hash21(cell + vec2(1.0, 0.0));
    float c = hash21(cell + vec2(0.0, 1.0));
    float d = hash21(cell + vec2(1.0, 1.0));

    return mix(
        mix(a, b, u.x),
        mix(c, d, u.x),
        u.y
    );
}

float fbm(vec2 p)
{
    float value = 0.0;
    float amplitude = 0.5;

    mat2 rotation = mat2(
         0.80, 0.60,
        -0.60, 0.80
    );

    for (int i = 0; i < 5; i++)
    {
        value += amplitude * noise2(p);
        p = rotation * p * 2.03 + vec2(13.1, 17.7);
        amplitude *= 0.5;
    }

    return value;
}

vec3 starLayer(
    vec2 uv,
    float gridSize,
    float layer,
    float pixelFootprint
)
{
    vec2 p = uv * gridSize;

    p += vec2(
        time * 0.035 * layer,
        time * 0.014 * layer
    );

    vec2 cell = floor(p);
    vec2 local = fract(p);

    float pixel = gridSize * pixelFootprint;
    vec3 color = vec3(0.0);

    // Include adjacent cells so stars do not clip at cell edges.
    for (int y = -1; y <= 1; y++)
    {
        for (int x = -1; x <= 1; x++)
        {
            vec2 offset = vec2(float(x), float(y));
            vec2 id = cell + offset;

            vec2 randomValue = hash22(
                id + vec2(layer * 37.0, layer * 19.0)
            );

            vec2 center =
                offset + vec2(0.15) + randomValue * 0.70;

            vec2 delta = local - center;

            float brightness = smoothstep(
                0.76,
                1.0,
                hash21(id + vec2(layer * 53.0, 8.0))
            );

            float radius =
                0.018 + 0.020 * pow(randomValue.x, 8.0);

            // Approximate pixel filtering with energy compensation.
            float widthSquared =
                radius * radius + pixel * pixel * 0.5;

            float star = exp(
                -dot(delta, delta) / widthSquared
            );

            star *= radius * radius / widthSquared;

            float twinkle =
                0.85 + 0.15 * sin(
                    time * (0.7 + 1.4 * randomValue.y) +
                    randomValue.x * TAU
                );

            vec3 starColor = mix(
                vec3(0.65, 0.80, 1.00),
                vec3(1.00, 0.73, 0.40),
                randomValue.y
            );

            color +=
                starColor *
                star *
                brightness *
                twinkle *
                8.0;
        }
    }

    return color;
}

void main()
{
    float mn = max(min(resolution.x, resolution.y), 1.0);
    vec2 uv = (gl_FragCoord.xy - 0.5 * resolution) / mn;

    float zoom = exp(
        clamp(wheel.y / mn, -3.0, 3.0) * 0.20
    );

    uv *= zoom;
    uv += move / mn * 0.25;

    vec2 p = uv * 3.0 + vec2(time * 0.012, 0.0);
    float t = time * 0.035;

    vec2 warp = vec2(
        fbm(p + vec2(0.0, t)),
        fbm(p + vec2(5.2, -t))
    );

    float cloud = fbm(
        p + warp * 3.0 + vec2(t * 0.2, 0.0)
    );

    float density = pow(
        smoothstep(0.28, 0.75, cloud),
        2.0
    );

    float cloudBand = exp(
        -pow(uv.y * 1.2 + uv.x * 0.25, 2.0)
    );

    vec3 cloudColor = mix(
        vec3(0.025, 0.040, 0.230),
        vec3(0.420, 0.035, 0.160),
        smoothstep(0.30, 0.70, warp.y)
    );

    vec3 color =
        cloudColor * density * cloudBand * 2.30;

    float brightCloud = pow(
        max(cloud - 0.45, 0.0),
        2.0
    );

    color +=
        vec3(0.07, 0.24, 0.45) *
        brightCloud *
        cloudBand *
        4.0;

    float pixelFootprint = zoom / mn;

    color += starLayer(uv, 18.0, 1.0, pixelFootprint);
    color += starLayer(uv, 34.0, 2.0, pixelFootprint) * 0.75;
    color += starLayer(uv, 58.0, 3.0, pixelFootprint) * 0.50;

    color = vec3(1.0) - exp(-color * 1.35);
    color = pow(max(color, vec3(0.0)), vec3(0.4545));

    O = vec4(color, 1.0);
}
```
